# Next.js Integration

This guide covers the recommended patterns for using the Watcher JS SDK in a Next.js application with the App Router.

## Installation

<PackageTabs pkg="@watcher/nextjs" />

## Create a singleton client

Create a shared client in `src/lib/watcher.ts` so you don't re-initialise on every request:

```typescript
// src/lib/watcher.ts
import { Watcher } from "@watcher24/js-sdk";

const globalForWatcher = globalThis as unknown as {
  watcher: Watcher | undefined;
};

export const watcher =
  globalForWatcher.watcher ??
  new Watcher({
    apiKey: process.env.WATCHER_API_KEY!,
    appId: "console",
    environment: process.env.NODE_ENV === "production" ? "production" : "development",
  });

if (process.env.NODE_ENV !== "production") {
  globalForWatcher.watcher = watcher;
}
```

## Server Components

```typescript
// app/dashboard/page.tsx
import { watcher } from "@/lib/watcher";

export default async function DashboardPage() {
  await watcher.log("dashboard.viewed", { severity: "info" });
  return <div>Dashboard</div>;
}
```

## Route Handlers

```typescript
// app/api/orders/route.ts
import { NextRequest, NextResponse } from "next/server";
import { watcher } from "@/lib/watcher";

export async function POST(req: NextRequest) {
  const body = await req.json();

  await watcher.audit({
    action: "order.created",
    actor: { id: body.userId },
    target: { type: "order", id: body.orderId },
  });

  return NextResponse.json({ ok: true });
}
```

## Middleware (request logging)

```typescript
// middleware.ts
import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";
import { watcher } from "@/lib/watcher";

export async function middleware(request: NextRequest) {
  const start = Date.now();
  const response = NextResponse.next();

  await watcher.log("http.request", {
    severity: "info",
    payload: {
      method: request.method,
      path: request.nextUrl.pathname,
      latency_ms: Date.now() - start,
      status: response.status,
    },
  });

  return response;
}

export const config = {
  matcher: ["/api/:path*"],
};
```

## Client-side logging

For browser-side events (user actions, UI errors), use the client directly:

```typescript
"use client";
import { Watcher } from "@watcher24/js-sdk/browser";

const watcher = new Watcher({
  apiKey: process.env.NEXT_PUBLIC_WATCHER_API_KEY!,
  appId: "console-browser",
});

export function TrackClick({ label }: { label: string }) {
  return (
    <button onClick={() => watcher.log(`click.${label}`, { severity: "debug" })}>
      {label}
    </button>
  );
}
```

<Callout type="warning" title="Public API key">
The browser SDK uses `NEXT_PUBLIC_WATCHER_API_KEY`. Create a separate, write-only API key for browser use — revoke it easily if it leaks.
</Callout>

## Environment variables

```bash title=".env.local"
# Server-only — never exposed to the browser
W24_API_KEY=wtch_server_...

# Browser-safe — create a separate write-only key
NEXT_PUBLIC_W24_API_KEY=wtch_pub_...
NEXT_PUBLIC_W24_GATEWAY_URL=https://ingest.watcher24.io
```
